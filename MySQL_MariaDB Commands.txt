* Table examples schema (database table names and structures)
DB      Name        Type           Size
--	----	    ----	   ----
scores:
	player_id - integer
	score	  - integer

players:
	id     	  - integer auto-increment primary key (index)
	name   	  - varchar - 32 (varchar is short for variable characters)
        handle 	  - alphanumeric - 16

* Create a new table and specify data using schema above for as example
create table <table name> ( <press enter to begin>
	-> id integer auto_increment primary key,
	-> name varchar(32) not null,
	-> handle varchar(16) not null );

* List all available tables
show tables;

* List available tables and their current contents
describe <table name>;

* Create data (insert)
insert into <table name> <press enter>
	-> (name, handle)
	-> values
	-> ("Paul", "The Boss");

* Read data (select) retrieve all the data in a table and display it
select * from <table name>;

* Read data (select) specify how many rows to retrieve
select * from <table name> limit <number>;

* Read specific data in particular order
select * from <table name>
	-> order by id desc (ascending or descending order)
	-> limit 2 (how many lines to order);

* Filter results from particular point
select * from <table name>
	-> where <data name> = <value>;

* Filter results from part of a name
select * from <table name> where <data name> like '<characters>%';

* Update a row
update <table name>
	-> set <data name> = <value>
	-> where <data name> = <value>;

* Log out
quit

* Pulling correlated info from separate tables
select players.name, players.handle, scores.score from players,
                                  -> scores where players.id = scores.player_id order by scores.score
                                  ->  desc

* Import data into a table from an external sql file
source <filename>.sql --special;

* Count the elements in a table
select count(*) from <table name>;

* Filter, compare, set shorthand
select c.name, r.name
      -> from countries as c, regions as r
      -> where c.region_id = r.region_id
      -> and r.name = "British Islands";

* Pull name and language from various tables using country and language id's
select c.name, l.language, cl.official
        -> from countries as c, languages as l, country_languages as cl
        -> where c.country_id = cl.country_id
        -> and cl.language_id = l.language_id
        -> and c.name = "India";
